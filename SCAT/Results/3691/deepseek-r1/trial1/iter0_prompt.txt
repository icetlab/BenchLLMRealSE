class Solution {
    public int minimumOperations(int[][] grid) {
        int count=0;
		for (int j=0;j<grid[0].length ;j++ )
		{
			for (int i=0;i<grid.length ;i++ )
			{
				for (int k=i+1;k<grid.length ;k++ )
				{
					while(grid[i][j]>=grid[k][j])
					{
						grid[k][j]++;
						count++;
					}
				}
			}
		}
		return count;
    }
}
