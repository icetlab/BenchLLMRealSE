    public Map<Integer, MyBean> findDistances(Map<Integer, List<MyBean>> adjMap, int source) {
        TreeSet<MyBean> ts = new TreeSet<>((b1, b2) -> {
            if (Long.compare(b1.distance, b2.distance) != 0) {
