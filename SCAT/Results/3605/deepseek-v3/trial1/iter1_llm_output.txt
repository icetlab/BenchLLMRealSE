class Solution {
    public int[] minBitwiseArray(List<Integer> nums) {
        int[] arr = new int[nums.size()];
        for (int i = 0; i < nums.size(); i++) {
            int n = nums.get(i);
            if ((n & 1) == 0) {
                arr[i] = -1;
                continue;
            }
            // Count trailing 1s
            int k = 0;
            int temp = n;
            while ((temp & 1) == 1) {
                k++;
                temp >>= 1;
            }
            arr[i] = n - (1 << (k - 1));
        }
        return arr;
    }
}
